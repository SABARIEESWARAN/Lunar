import datetime
from config import MAX_APPOINTMENTS_PER_DAY, APPOINTMENT_SLOT_MINUTES, APPOINTMENT_TYPES

def schedule_appointment(
    patient_id,
    doctor_id,
    appointment_date,
    appointment_type,
    patient_age,
    notes=None,
    priority=None,
    insurance_id=None,
    referring_doctor=None,
):
    """
    Schedule a new appointment for a patient.

    SEEDED ISSUE #3 — High complexity function:
    This function is too long (80+ lines) and has too many branches (12+).
    It handles scheduling logic, validation, conflict detection, insurance checks,
    priority queuing, and notification in a single function.
    """
    # Input validation block
    if not patient_id:
        raise ValueError("patient_id is required")
    if not doctor_id:
        raise ValueError("doctor_id is required")
    if not appointment_date:
        raise ValueError("appointment_date is required")
    if appointment_type not in APPOINTMENT_TYPES:
        raise ValueError(f"Unknown appointment type: {appointment_type}")

    # Validate patient age for appointment type
    if not validate_patient_age(patient_age):
        raise ValueError(f"Invalid patient age: {patient_age}")

    # Pediatric appointments need special handling
    if patient_age < 18:
        if appointment_type == "specialist_referral" and not referring_doctor:
            raise ValueError("Pediatric specialist referrals require a referring doctor")
        if appointment_type == "emergency":
            priority = "urgent"

    # Date validation
    if isinstance(appointment_date, str):
        try:
            appointment_date = datetime.datetime.fromisoformat(appointment_date)
        except ValueError:
            raise ValueError(f"Invalid date format: {appointment_date}")

    if appointment_date < datetime.datetime.now():
        raise ValueError("Cannot schedule appointments in the past")

    # Check daily limit
    existing = _get_appointments_for_date(doctor_id, appointment_date.date())
    if len(existing) >= MAX_APPOINTMENTS_PER_DAY:
        raise ValueError(f"Doctor {doctor_id} has reached the daily appointment limit")

    # Check for scheduling conflicts
    for existing_appt in existing:
        existing_time = existing_appt.get("time")
        if existing_time:
            diff = abs((appointment_date - existing_time).total_seconds() / 60)
            if diff < APPOINTMENT_SLOT_MINUTES:
                raise ValueError(
                    f"Scheduling conflict: appointment within {APPOINTMENT_SLOT_MINUTES} minutes of another"
                )

    # Insurance validation
    if insurance_id:
        if not _validate_insurance(insurance_id, appointment_type):
            if priority == "urgent":
                # Allow urgent appointments without valid insurance but flag it
                notes = (notes or "") + " [INSURANCE_UNVERIFIED]"
            else:
                raise ValueError(f"Insurance {insurance_id} does not cover {appointment_type}")

    # Set priority based on appointment type if not explicitly set
    if priority is None:
        if appointment_type == "emergency":
            priority = "urgent"
        elif appointment_type == "follow_up":
            priority = "normal"
        elif appointment_type == "specialist_referral":
            priority = "elevated"
        else:
            priority = "normal"

    # Build appointment record
    appointment = {
        "patient_id": patient_id,
        "doctor_id": doctor_id,
        "datetime": appointment_date.isoformat(),
        "type": appointment_type,
        "priority": priority,
        "notes": notes,
        "insurance_id": insurance_id,
        "referring_doctor": referring_doctor,
        "status": "scheduled",
    }

    _save_appointment(appointment)

    # Send notifications
    if priority == "urgent":
        _notify_urgent(patient_id, doctor_id, appointment_date)
    else:
        _notify_standard(patient_id, appointment_date)

    return appointment

"""

DEMO REPOSITORY — all data is fictional.

from config import APPOINTMENT_TYPES


PROCEDURE_COSTS = {
    "general_consultation": 150.00,
    "follow_up": 75.00,
    "emergency": 500.00,
    "specialist_referral": 300.00,
    "lab_results": 50.00,
}

INSURANCE_DISCOUNT_RATES = {
    "BASIC": 0.20,
    "STANDARD": 0.40,
    "PREMIUM": 0.60,
}


def process_billing(patient_id, appointment_id, appointment_type, insurance_plan=None, discount_code=None):
    """
    Process billing for a completed appointment.

    SEEDED ISSUE #4 — Deeply nested conditional logic
    This function has 4+ levels of nesting and mixes billing logic,
    discount calculation, insurance processing, and persistence.
    """
    if patient_id:
        if appointment_type in PROCEDURE_COSTS:
            base_cost = PROCEDURE_COSTS[appointment_type]
            final_cost = base_cost

            if insurance_plan:
                if insurance_plan.upper() in INSURANCE_DISCOUNT_RATES:
                    discount = INSURANCE_DISCOUNT_RATES[insurance_plan.upper()]
                    final_cost = base_cost * (1 - discount)
                    if discount_code:
                        if _validate_discount_code(discount_code):
                            if discount_code.startswith("VIP"):
                                final_cost = final_cost * 0.85
                            elif discount_code.startswith("SENIOR"):
                                final_cost = final_cost * 0.90
                            else:
                                final_cost = final_cost * 0.95
                else:
                    # Unknown insurance plan — charge full price
                    final_cost = base_cost
            else:
                if discount_code:
                    if _validate_discount_code(discount_code):
                        final_cost = base_cost * 0.95

            invoice = {
                "patient_id": patient_id,
                "appointment_id": appointment_id,
                "appointment_type": appointment_type,
                "base_cost": base_cost,
                "final_cost": round(final_cost, 2),
                "insurance_plan": insurance_plan,
                "discount_code": discount_code,
                "status": "pending",
            }

            # SEEDED ISSUE #6 — weak error handling around DB write
            _save_invoice(invoice)
            return invoice
        else:
            raise ValueError(f"Unknown appointment type for billing: {appointment_type}")
    else:
        raise ValueError("patient_id is required")

    if age < 0 or age > 150:

def validate_patient_id(patient_id):
    """Check that a patient ID matches the required format."""
    if not patient_id:
        return False
    if not isinstance(patient_id, str):
        return False
    # Patient IDs must be 8 alphanumeric characters
    import re
    return bool(re.match(r"^[A-Z]{2}\d{6}$", patient_id))


def register_patient(name, age, patient_id, contact_email, blood_type=None):
    """Register a new patient in the system.

    Returns a patient record dict or raises ValueError on invalid input.
    """
    if not name or not isinstance(name, str):
        raise ValueError("Patient name is required")

    if not validate_patient_id(patient_id):
        raise ValueError(f"Invalid patient ID format: {patient_id}")

        "patient_id": patient_id,
        "name": name,
        "age": age,
        "contact_email": contact_email,
        "blood_type": blood_type,
        "status": "active",
    }
