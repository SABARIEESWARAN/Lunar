import React, { useState, useEffect, useRef, useCallback } from "react";
import { createRoot } from "react-dom/client";
import "./styles.css";

const API = import.meta.env.VITE_API_URL || "";

// ─── API helpers ──────────────────────────────────────────────────────────────

async function apiFetch(path, opts = {}) {
  const res = await fetch(API + path, {
    headers: { "Content-Type": "application/json" },
    ...opts,
  });
  const json = await res.json();
  if (!res.ok) throw new Error(json.detail || `HTTP ${res.status}`);
  return json;
}

// ─── App root ─────────────────────────────────────────────────────────────────

function App() {
  const [url, setUrl] = useState("demo://hospital");
  const [jobId, setJobId] = useState(null);
  const [job, setJob] = useState(null);   // full job status object
  const [loading, setLoading] = useState(false);
  const [view, setView] = useState("overview"); // overview | findings | agents | repair | verify
  const [selectedFinding, setSelectedFinding] = useState(null);
  const [repair, setRepair] = useState(null);      // current repair proposal
  const [verification, setVerification] = useState(null);
  const [diffData, setDiffData] = useState(null);
  const [actionLoading, setActionLoading] = useState(false);
  const [actionError, setActionError] = useState(null);
  const [messages, setMessages] = useState([
    { role: "ai", text: "I'm LUNAR. Give me a public GitHub repository and I'll map its maintenance risks." },
  ]);
  const [chat, setChat] = useState("");
  const pollRef = useRef(null);

  // ── Poll job status ──────────────────────────────────────────────────────
  const startPolling = useCallback((id) => {
    if (pollRef.current) clearInterval(pollRef.current);
    pollRef.current = setInterval(async () => {
      try {
        const j = await apiFetch(`/api/analyze/${id}`);
        setJob(j);
        if (j.status === "completed" || j.status === "failed") {
          clearInterval(pollRef.current);
          pollRef.current = null;
          setLoading(false);
          if (j.status === "failed") {
            setMessages((m) => [...m, { role: "ai", text: `Analysis failed: ${j.error}` }]);
          }
        }
      } catch (e) {
        clearInterval(pollRef.current);
        setLoading(false);
      }
    }, 1500);
  }, []);

  useEffect(() => () => { if (pollRef.current) clearInterval(pollRef.current); }, []);

  async function analyze() {
    if (!url.trim()) return;
    setLoading(true);
    setJob(null);
    setJobId(null);
    setSelectedFinding(null);
    setRepair(null);
    setVerification(null);
    setDiffData(null);
    setActionError(null);
    setView("overview");
    try {
      const res = await apiFetch("/api/analyze", {
        method: "POST",
        body: JSON.stringify({ repo_url: url }),
      });
      setJobId(res.job_id);
      setJob({ job_id: res.job_id, status: "queued", progress: 0, stage: "queued", result: null });
      startPolling(res.job_id);
      setMessages((m) => [...m, { role: "user", text: `Analyze: ${url}` }, { role: "ai", text: "Analysis started. Running parallel agents…" }]);
    } catch (e) {
      setLoading(false);
      setMessages((m) => [...m, { role: "ai", text: e.message }]);
    }
  }

  // ── Repair actions ────────────────────────────────────────────────────────

  async function prepareFix(finding) {
    setActionLoading(true);
    setActionError(null);
    setRepair(null);
    setDiffData(null);
    setVerification(null);
    try {
      const r = await apiFetch("/api/repair", {
        method: "POST",
        body: JSON.stringify({ job_id: jobId, finding_id: finding.id }),
      });
      setRepair(r);
      setView("repair");
      if (r.status === "manual_review_required") {
        setMessages((m) => [...m, { role: "ai", text: `Manual review required: ${r.reason}` }]);
      } else {
        setMessages((m) => [...m, { role: "ai", text: `Repair proposed for ${finding.title}. Review the diff before approving.` }]);
      }
    } catch (e) {
      setActionError(e.message);
    } finally {
      setActionLoading(false);
    }
  }

  async function viewDiff() {
    if (!repair?.repair_id) return;
    setActionLoading(true);
    setActionError(null);
    try {
      const d = await apiFetch(`/api/repair/${repair.repair_id}/diff`);
      setDiffData(d);
    } catch (e) {
      setActionError(e.message);
    } finally {
      setActionLoading(false);
    }
  }

  async function approveRepair() {
    if (!repair?.repair_id) return;
    setActionLoading(true);
    setActionError(null);
    try {
      const r = await apiFetch(`/api/repair/${repair.repair_id}/approve`, { method: "POST" });
      setRepair((prev) => ({ ...prev, status: r.status }));
      setMessages((m) => [...m, { role: "ai", text: "Repair approved. Run verification to confirm the change is safe." }]);
    } catch (e) {
      setActionError(e.message);
    } finally {
      setActionLoading(false);
    }
  }

  async function rejectRepair() {
    if (!repair?.repair_id) return;
    setActionLoading(true);
    setActionError(null);
    try {
      const r = await apiFetch(`/api/repair/${repair.repair_id}/reject`, { method: "POST" });
      setRepair((prev) => ({ ...prev, status: r.status }));
      setMessages((m) => [...m, { role: "ai", text: "Repair rejected and workspace cleaned up." }]);
    } catch (e) {
      setActionError(e.message);
    } finally {
      setActionLoading(false);
    }
  }

  async function runVerification() {
    if (!repair?.repair_id) return;
    setActionLoading(true);
    setActionError(null);
    try {
      const v = await apiFetch(`/api/verify/${repair.repair_id}`, { method: "POST" });
      setVerification(v);
      setRepair((prev) => ({ ...prev, status: v.status === "passed" ? "verified" : "verification_failed" }));
      setView("verify");
      setMessages((m) => [...m, { role: "ai", text: v.summary }]);
    } catch (e) {
      setActionError(e.message);
    } finally {
      setActionLoading(false);
    }
  }

  // ── Chat ─────────────────────────────────────────────────────────────────

  function ask() {
    if (!chat.trim()) return;
    const userMsg = chat;
    setChat("");
    let aiReply = "Analysis context is not loaded yet. Start an analysis first.";
    if (job?.result) {
      const data = job.result;
      if (userMsg.toLowerCase().includes("critical")) {
        const criticals = data.findings.filter((f) => f.severity === "critical");
        aiReply = criticals.length > 0
          ? `${criticals.length} critical finding(s): ${criticals.map((f) => f.title).join(", ")}.`
          : "No critical findings detected.";
      } else if (userMsg.toLowerCase().includes("health")) {
        aiReply = `Health score: ${data.health_score}/100. Technical debt: ${data.technical_debt_hours}h estimated.`;
      } else if (userMsg.toLowerCase().includes("fix") || userMsg.toLowerCase().includes("repair")) {
        aiReply = selectedFinding
          ? `Select 'PREPARE FIX' on the finding panel to generate a repair proposal for: ${selectedFinding.title}.`
          : "Select a finding from the Priority Findings list, then use PREPARE FIX.";
      } else {
        aiReply = `Repository scanned: ${data.repository.files} files, ${data.findings.length} findings, health ${data.health_score}/100, debt ${data.technical_debt_hours}h.`;
      }
    }
    setMessages((m) => [...m, { role: "user", text: userMsg }, { role: "ai", text: aiReply }]);
  }

  const data = job?.result;

  return (
    <div className="app">
      <aside>
        <div className="logo">LUNAR<span>●</span></div>
        <div className="tag">AUTONOMOUS MAINTENANCE</div>
        <nav>
          <b onClick={() => setView("overview")} style={{ cursor: "pointer" }}>Overview</b>
          <span onClick={() => setView("findings")} style={{ cursor: "pointer", background: view === "findings" ? "#151e2c" : undefined, color: view === "findings" ? "#fff" : undefined }}>Findings</span>
          <span onClick={() => setView("agents")} style={{ cursor: "pointer", background: view === "agents" ? "#151e2c" : undefined, color: view === "agents" ? "#fff" : undefined }}>Agents</span>
          {repair && <span onClick={() => setView("repair")} style={{ cursor: "pointer", background: view === "repair" ? "#151e2c" : undefined, color: view === "repair" ? "#fff" : undefined }}>Repair</span>}
          {verification && <span onClick={() => setView("verify")} style={{ cursor: "pointer", background: view === "verify" ? "#151e2c" : undefined, color: view === "verify" ? "#fff" : undefined }}>Verification</span>}
        </nav>
        <div className="sidecard">
          <small>WORKSPACE</small>
          <strong>Hospital Demo</strong>
          <span>Public repository</span>
        </div>
      </aside>

      <main>
        <header>
          <div>
            <small>REPOSITORY / ANALYSIS</small>
            <h1>Codebase Health</h1>
          </div>
          <div className="live" style={{ color: loading ? "#78a7ff" : data ? "#73d6a2" : "#73d6a2" }}>
            ● {loading ? "ANALYZING" : data ? "ANALYSIS COMPLETE" : "SYSTEM READY"}
          </div>
        </header>

        <section className="repo">
          <input
            value={url}
            onChange={(e) => setUrl(e.target.value)}
            onKeyDown={(e) => e.key === "Enter" && !loading && analyze()}
            placeholder="https://github.com/owner/public-repository"
          />
          <button onClick={analyze} disabled={loading || !url.trim()}>
            {loading ? "ANALYZING…" : "START ANALYSIS"}
          </button>
          <button className="secondary" onClick={() => { setUrl("demo://hospital"); }} disabled={loading}>
            LOAD HOSPITAL DEMO
          </button>
        </section>

        {/* Progress bar during analysis */}
        {loading && job && (
          <div style={{ margin: "14px 0" }}>
            <div style={{ display: "flex", justifyContent: "space-between", fontSize: 11, color: "#66758a", marginBottom: 6 }}>
              <span>{STAGE_LABELS[job.stage] || job.stage}</span>
              <span>{job.progress}%</span>
            </div>
            <div style={{ background: "#111a28", borderRadius: 6, height: 6 }}>
              <div style={{ background: "#4d7cff", height: 6, borderRadius: 6, width: `${job.progress}%`, transition: "width 0.4s ease" }} />
            </div>
          </div>

// ─── Diff Viewer ──────────────────────────────────────────────────────────────

function DiffViewer({ diff }) {
  if (!diff) {
    return <p style={{ color: "#66758a", fontSize: 13, padding: "8px 0" }}>No diff available (file may be new or unchanged).</p>;
  }
  return (
    <pre style={{
      background: "#070b12",
      border: "1px solid #1b2534",
      borderRadius: 8,
      padding: 14,
      overflowX: "auto",
      fontSize: 12,
      lineHeight: 1.7,
      fontFamily: '"SFMono-Regular", Consolas, monospace',
      margin: 0,
    }}>
      {diff.split("\n").map((line, i) => {
        let color = "#8d9aad";
        if (line.startsWith("+") && !line.startsWith("+++")) color = "#73d6a2";
        else if (line.startsWith("-") && !line.startsWith("---")) color = "#f87171";
        else if (line.startsWith("@@")) color = "#78a7ff";
        else if (line.startsWith("---") || line.startsWith("+++")) color = "#dce7f5";
        return <span key={i} style={{ color, display: "block" }}>{line}</span>;
      })}
    </pre>
  );
}

function MetricCompare({ label, before, after, unit = "", higherBetter }) {
  const delta = (after ?? 0) - (before ?? 0);
  const improved = higherBetter ? delta >= 0 : delta <= 0;
  const deltaStr = delta === 0 ? "±0" : (delta > 0 ? `+${delta}` : `${delta}`);
  return (
    <div style={{ background: "#0d131e", border: "1px solid #1b2534", borderRadius: 8, padding: "10px 14px" }}>
      <div style={{ fontSize: 10, color: "#66758a", textTransform: "uppercase", letterSpacing: ".5px", marginBottom: 6 }}>{label}</div>
      <div style={{ display: "flex", gap: 10, alignItems: "baseline" }}>
        <span style={{ fontSize: 13, color: "#8d9aad" }}>{before}{unit}</span>
        <span style={{ fontSize: 11, color: "#66758a" }}>→</span>
        <span style={{ fontSize: 18, fontWeight: 700, color: "#dce7f5" }}>{after}{unit}</span>
        <span style={{ fontSize: 12, color: improved ? "#73d6a2" : "#f87171", fontWeight: 700 }}>{deltaStr}</span>
      </div>
    </div>
  );
}

}
